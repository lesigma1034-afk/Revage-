local friendCache = {}
local function IsFriendCached(userId)
    if not userId then return false end
    if friendCache[userId] ~= nil then return friendCache[userId] end
    task.spawn(function()
        local f = false
        pcall(function() f = LocalPlayer:IsFriendsWith(userId) end)
        friendCache[userId] = f
    end)
    return false
end

local function IsWhitelisted(player)
    if not player then return false end
    if Settings.Whitelist.Players[player.Name]
       or Settings.Whitelist.Players[player.UserId]
       or Settings.Whitelist.Players[tostring(player.UserId)]
       or Settings.Whitelist.Players[string.lower(player.Name)] then
        return true
    end
    if Settings.Whitelist.Team and LocalPlayer.Team and player.Team and LocalPlayer.Team == player.Team then
        return true
    end
    if Settings.Whitelist.Friends and IsFriendCached(player.UserId) then return true end
    return false
end

local function IsValidTarget(player)
    if player == LocalPlayer or not player.Character then return false end
    if IsWhitelisted(player) then return false end
    local char = player.Character
    local humanoid = char:FindFirstChild("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not humanoid or humanoid.Health <= 0 or not hrp then return false end
    local bodyEffects = char:FindFirstChild("BodyEffects")
    if bodyEffects then
        local ko = bodyEffects:FindFirstChild("K.O") or bodyEffects:FindFirstChild("KO")
        if ko and ko.Value == true then return false end
    end
    return true
end

local function GetCharacterAimPart(character, chosenPart)
    if not character then return nil end
    local partName = chosenPart or "HumanoidRootPart"
    if partName == "Random" then
        local parts = {"Head", "UpperTorso", "HumanoidRootPart"}
        partName = parts[math.random(1, #parts)]
    end
    local part = character:FindFirstChild(partName)
    if not part and (partName == "Head" or partName == "UpperTorso") then
        part = character:FindFirstChild("Torso") or character:FindFirstChild("HumanoidRootPart")
    end
    return part or character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Head")
end

-- жёсткий prediction: бег + прыжок / макро
local function GetPredictedPosition(part, usePred, predAmount)
    if not part then return nil end
    local pos = part.Position
    if not usePred then return pos end

    local char = part.Parent
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local vel = (hrp and (hrp.AssemblyLinearVelocity or hrp.Velocity))
             or part.AssemblyLinearVelocity
             or part.Velocity
             or Vector3.zero

    local amount = predAmount or 0.165
    local speed = vel.Magnitude

    -- ускорение prediction на быстром стрейфе
    if speed > 28 then
        amount = amount * 1.35
    elseif speed > 18 then
        amount = amount * 1.15
    end

    -- в воздухе / прыжок — сильнее по Y
    local airborne = false
    if hum then
        local st = hum:GetState()
        if st == Enum.HumanoidStateType.Freefall
        or st == Enum.HumanoidStateType.Jumping
        or st == Enum.HumanoidStateType.FallingDown then
            airborne = true
        end
        if hum.FloorMaterial == Enum.Material.Air then
            airborne = true
        end
    end

    local predVel = vel
    if airborne then
        -- макро-прыжки: чуть вперёд по горизонтали + вертикаль
        predVel = Vector3.new(vel.X * 1.1, vel.Y * 1.25, vel.Z * 1.1)
        amount = amount * 1.2
    end

    return pos + predVel * amount
end

local function GetClosestPlayerForAim()
    local cam = workspace.CurrentCamera or Camera
    if not cam then return nil end
    local closest, shortestDist = nil, Settings.Aimbot.FOV or 220
    local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
    for _, player in ipairs(Players:GetPlayers()) do
        if IsValidTarget(player) then
            local part = GetCharacterAimPart(player.Character, Settings.Aimbot.AimPart)
            if part then
                local screenPos, onScreen = cam:WorldToViewportPoint(part.Position)
                if onScreen and screenPos.Z > 0 then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                    if dist <= shortestDist then
                        shortestDist = dist
                        closest = player
                    end
                end
            end
        end
    end
    return closest
end

-- REAL HitChance: 100 = всегда, иначе roll 1-100
local function RollHitChance()
    local hc = Settings.SilentAim.HitChance or 100
    if hc >= 100 then return true end
    if hc <= 0 then return false end
    return math.random(1, 100) <= hc
end

local function CreateCycle(parent, label, options, defaultIndex, callback)

    local btn = Instance.new("TextButton", parent)

    btn.Size = UDim2.new(1, 0, 0, 34)
    btn.Font = Enum.Font.GothamSemibold
    btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
    btn.TextColor3 = Color3.fromRGB(210, 190, 255)
    btn.TextSize = 11
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 10)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = Color3.fromRGB(100, 45, 170)
    stroke.Thickness = 1.3
    btn.MouseButton1Click:Connect(function()
        idx = idx + 1
        if idx > #options then idx = 1 end
        btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
        callback(options[idx])
    end)
    return {
        Button = btn,
        Get = function() return options[idx] end,
        Set = function(self, val)
            for i, opt in ipairs(options) do
                if opt == val then
                    idx = i
                    btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
                    callback(options[idx])
                    break
                end
            end
        end
    }
end

local function DoVClip(distance)
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum or hum.Health <= 0 then return end
    hrp.CFrame = hrp.CFrame + Vector3.new(0, distance, 0)
    hrp.AssemblyLinearVelocity = Vector3.zero
    if currentFlyCFrame then currentFlyCFrame = currentFlyCFrame + Vector3.new(0, distance, 0) end
end

local function RestoreFlyCollision(char)
    if not char then return end
    for part, can in pairs(flyPartsCanCollide) do
        if part and part.Parent then pcall(function() part.CanCollide = can end) end
    end
    flyPartsCanCollide = {}
end

local function UpdateFlyState(state, fromUI)
    Settings.Misc.Fly = state
    if state then
        FlyFloatBtn.BackgroundColor3 = Color3.fromRGB(210, 65, 35)
        FlyFloatBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
        FlyFloatBtn.Text = "🚀 FLY: ON"
        flyFloatStroke.Color = Color3.fromRGB(255, 130, 70)
    else
        FlyFloatBtn.BackgroundColor3 = Color3.fromRGB(22, 16, 20)
        FlyFloatBtn.TextColor3 = Color3.fromRGB(240, 180, 160)
        FlyFloatBtn.Text = "🚀 FLY: OFF"
        flyFloatStroke.Color = Color3.fromRGB(170, 70, 45)
    end
    if not fromUI and flyToggleObj and flyToggleObj.Set then flyToggleObj:Set(state) end

    if state then
        if flyConnection then return end
        local startHrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if startHrp then currentFlyCFrame = startHrp.CFrame end
        flyPartsCanCollide = {}
        flyConnection = RunService.Heartbeat:Connect(function(dt)
            if not Settings.Misc.Fly then return end
            local char = LocalPlayer.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if not hrp or not hum or hum.Health <= 0 then return end
            local cam = workspace.CurrentCamera or Camera
            if not cam then return end
            local camCF = cam.CFrame
            if not currentFlyCFrame then currentFlyCFrame = hrp.CFrame end
            for _, part in ipairs(char:GetChildren()) do
                if part:IsA("BasePart") then
                    if flyPartsCanCollide[part] == nil then flyPartsCanCollide[part] = part.CanCollide end
                    if part.CanCollide then part.CanCollide = false end
                end
            end
            hrp.AssemblyLinearVelocity = Vector3.zero
            local moveVector = Vector3.zero
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveVector = moveVector + camCF.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveVector = moveVector - camCF.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveVector = moveVector - camCF.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveVector = moveVector + camCF.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveVector = moveVector + Vector3.new(0, 1, 0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) or UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
                moveVector = moveVector - Vector3.new(0, 1, 0)
            end
            if moveVector.Magnitude == 0 and hum.MoveDirection.Magnitude > 0 then
                local md = hum.MoveDirection
                moveVector = moveVector + Vector3.new(md.X, camCF.LookVector.Y * md.Magnitude, md.Z)
            end
            if moveVector.Magnitude > 0 then
                currentFlyCFrame = currentFlyCFrame + moveVector.Unit * ((Settings.Misc.FlySpeed or 50) * (dt or 0.016))
            end
            hrp.CFrame = CFrame.new(currentFlyCFrame.Position) * camCF.Rotation
            hrp.AssemblyLinearVelocity = Vector3.zero
        end)
        table.insert(_G.CociteConnections, flyConnection)
    else
        if flyConnection then flyConnection:Disconnect() flyConnection = nil end
        currentFlyCFrame = nil
        RestoreFlyCollision(LocalPlayer.Character)
        local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if hrp then hrp.AssemblyLinearVelocity = Vector3.zero end
    end
end

local function ToggleSilentLock()
    if not Settings.SilentAim.Enabled then return end
    if lockedSilentTarget then
        lockedSilentTarget = nil
        SilentFloatBtn.BackgroundColor3 = Color3.fromRGB(0, 140, 190)
        SilentFloatBtn.Text = "🔮 SILENT: ON"
    else
        local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
        local shortestDist, bestTarget = Settings.SilentAim.FOV, nil
        for _, player in ipairs(Players:GetPlayers()) do
            if IsValidTarget(player) then
                local part = GetCharacterAimPart(player.Character, Settings.SilentAim.AimPart)
                if part then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
                    if onScreen and screenPos.Z > 0 then
                        local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                        if dist <= shortestDist then
                            if not Settings.SilentAim.WallCheck or IsWallCheckVisible(part, player.Character) then
                                shortestDist = dist
                                bestTarget = player
                            end
                        end
                    end
                end
            end
        end
        if bestTarget then
            lockedSilentTarget = bestTarget
            SilentFloatBtn.BackgroundColor3 = Color3.fromRGB(0, 210, 110)
            SilentFloatBtn.Text = "🔮 LOCKED: " .. bestTarget.Name
            if Settings.TargetStrafe.Enabled then targetStrafeTarget = bestTarget end
        end
    end
end

IsWallCheckVisible = function(part, targetChar)
    if not Settings.SilentAim.WallCheck then return true end
    local localChar = LocalPlayer.Character
    if not localChar then return false end
    local origin = Camera.CFrame.Position
    local dir = part.Position - origin
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {localChar, Camera}
    params.IgnoreWater = true
    isInternalRay = true
    local hit = workspace:Raycast(origin, dir, params)
    isInternalRay = false
    return (not hit) or hit.Instance:IsDescendantOf(targetChar)
end

local function CalcRayDirection(origin, targetPos, spread)
    local dir = (targetPos - origin).Unit
    if spread and spread > 0 then
        local spreadDist = spread * ((targetPos - origin).Magnitude / 100)
        dir = ((targetPos + Vector3.new(
            math.random(-spreadDist * 1000, spreadDist * 1000) / 1000,
            math.random(-spreadDist * 1000, spreadDist * 1000) / 1000,
            math.random(-spreadDist * 1000, spreadDist * 1000) / 1000
        )) - origin).Unit
    end
    return dir * 1000
end

RunService.RenderStepped:Connect(function(dt)
    if not AUTHED or not Settings.TargetStrafe.Enabled then return end
    if not targetStrafeTarget or not IsValidTarget(targetStrafeTarget) then
        if lockedSilentTarget and IsValidTarget(lockedSilentTarget) then
            targetStrafeTarget = lockedSilentTarget
        else
            targetStrafeTarget = GetClosestPlayerForAim()
        end
        if not targetStrafeTarget then
            UpdateStrafeState(false, false)
            return
        end
    end
    local char = LocalPlayer.Character
    local tChar = targetStrafeTarget.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end
    if not tChar or not tChar:FindFirstChild("HumanoidRootPart") then return end
    local hrp, thrp = char.HumanoidRootPart, tChar.HumanoidRootPart
    local hum = char:FindFirstChildOfClass("Humanoid")
    local thum = tChar:FindFirstChildOfClass("Humanoid")
    if (hum and hum.Health <= 0) or (thum and thum.Health <= 0) then
        UpdateStrafeState(false, false)
        return
    end
    local ts = Settings.TargetStrafe
    strafeAngle = strafeAngle + (ts.Speed * dt)
    if ts.Mode == "Orbit" then
        local x = thrp.Position.X + math.cos(strafeAngle) * ts.Radius
        local z = thrp.Position.Z + math.sin(strafeAngle) * ts.Radius
        hrp.CFrame = CFrame.new(Vector3.new(x, thrp.Position.Y + 3, z), thrp.Position)
    elseif ts.Mode == "Above" then
        local x = thrp.Position.X + math.cos(strafeAngle) * (ts.Radius / 2)
        local z = thrp.Position.Z + math.sin(strafeAngle) * (ts.Radius / 2)
        hrp.CFrame = CFrame.new(Vector3.new(x, thrp.Position.Y + ts.Height, z), thrp.Position)
    else
        local x = thrp.Position.X + math.cos(strafeAngle) * ts.Radius + math.random(-5, 5)
        local z = thrp.Position.Z + math.sin(strafeAngle) * ts.Radius + math.random(-5, 5)
        hrp.CFrame = CFrame.new(Vector3.new(x, thrp.Position.Y + ts.Height, z), thrp.Position)
    end
end)

        local oldIndex
        oldIndex = hookmetamethod(game, "__index", newcclosure(function(self, key)
            if not checkcaller() and AUTHED and self == Mouse then
                if Settings.SilentAim.Enabled and currentSilentAimPos then
                    if key == "Hit" or key == "hit" then return CFrame.new(currentSilentAimPos) end
                    if (key == "Target" or key == "target") and currentSilentTargetPart then return currentSilentTargetPart end
                end
            end
            return oldIndex(self, key)
        end))
    end
end)

pcall(function()
    if hookmetamethod and getnamecallmethod and newcclosure and checkcaller then
        local oldNamecall
        oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(...)
            local method = getnamecallmethod()
            local args = {...}
            local self = args[1]
            -- тянем ТОЛЬКО если currentSilentAimPos есть (после успешного HitChance)
            if not checkcaller() and AUTHED and not isInternalRay and Settings.SilentAim.Enabled and currentSilentAimPos then
                if self == workspace and method == "Raycast" then
                    if typeof(args[2]) == "Vector3" and typeof(args[3]) == "Vector3" then
                        local toolName = equippedToolName or ""
                        local isShotgun = string.find(string.lower(toolName), "double")
                                       or string.find(string.lower(toolName), "tactical")
                                       or string.find(string.lower(toolName), "shotgun")
                        local spreadVal = 0
                        if not (Settings.NoSpread and Settings.NoSpread.Enabled) then
                            spreadVal = isShotgun and 6 or 0
                        else
                            spreadVal = ((Settings.NoSpread.Amount or 0) / 100) * (isShotgun and 6 or 0)
                        end
                        args[3] = CalcRayDirection(args[2], currentSilentAimPos, spreadVal)
                        return oldNamecall(table.unpack(args))
                    end
                elseif method == "FireServer" and tostring(self) == "MainEvent" then
                    local eventArg = tostring(args[2] or args[1] or "")
                    if eventArg == "UpdateMousePos" or eventArg == "UpdateMousePosI" or string.find(string.lower(eventArg), "mousepos") then
                        if args[2] == eventArg then args[3] = currentSilentAimPos
                        else args[2] = currentSilentAimPos end
                        return oldNamecall(table.unpack(args))
                    end
                end
            end
            return oldNamecall(...)
        end))
    end
end)

RunService:BindToRenderStep("CociteCamLock", Enum.RenderPriority.Camera.Value + 1, function(dt)
    pcall(function()
        if not AUTHED or not Settings.Aimbot.Enabled or not AimActive then return end
        local cam = workspace.CurrentCamera or Camera
        if not cam then return end
        local target = Settings.Aimbot.LockedTarget
        if target and not IsValidTarget(target) then
            target = nil
            Settings.Aimbot.LockedTarget = nil
        end
        if not target then
            target = GetClosestPlayerForAim()
            Settings.Aimbot.LockedTarget = target
        end
        if target and target.Character then
            local part = GetCharacterAimPart(target.Character, Settings.Aimbot.AimPart)
            if part then
                local aimPos = GetPredictedPosition(part, Settings.Aimbot.UsePrediction, Settings.Aimbot.Prediction)
                if aimPos then
                    local targetCF = CFrame.lookAt(cam.CFrame.Position, aimPos)
                    if Settings.Aimbot.Smooth then
                        local sm = math.clamp(Settings.Aimbot.Smoothness or 0.35, 0.01, 1)
                        local sf = math.clamp(1 - math.pow(1 - math.clamp(sm, 0.01, 0.99), (dt or 0.016) * 60), 0.01, 1)
                        cam.CFrame = cam.CFrame:Lerp(targetCF, sf)
                    else
                        cam.CFrame = targetCF
                    end
                end
            end
        end
    end)
end)

local function RemoveHighlight(player)
    if Settings.ESP.ActiveHighlights[player] then
        Settings.ESP.ActiveHighlights[player]:Destroy()
        Settings.ESP.ActiveHighlights[player] = nil
    end
end
local function ClearESP()
    for _, list in pairs(Settings.ESP.ActiveDrawings) do
        for _, o in pairs(list) do if o then o:Remove() end end
    end
    Settings.ESP.ActiveDrawings = {}
    for p, _ in pairs(Settings.ESP.ActiveHighlights) do RemoveHighlight(p) end
end

Players.PlayerRemoving:Connect(function(player)
    if Settings.ESP.ActiveDrawings[player] then
        for _, o in pairs(Settings.ESP.ActiveDrawings[player]) do if o then o:Remove() end end
        Settings.ESP.ActiveDrawings[player] = nil
    end
    RemoveHighlight(player)
    if lockedSilentTarget == player then lockedSilentTarget = nil end
    if targetStrafeTarget == player then targetStrafeTarget = nil end
end)

RunService.RenderStepped:Connect(function()
    if not AUTHED then return end
    for _, player in ipairs(Players:GetPlayers()) do
        if Settings.ESP.Enabled and IsValidTarget(player) then
            local char = player.Character
            local hrp = char:FindFirstChild("HumanoidRootPart")
            local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
            if not Settings.ESP.ActiveDrawings[player] then
                local box = Drawing.new("Square")
                box.Visible = false
                box.Color = Color3.fromRGB(180, 50, 255)
                box.Thickness = 1.5
                box.Filled = false
                local nameText = Drawing.new("Text")
                nameText.Visible = false
                nameText.Color = Color3.fromRGB(255, 255, 255)
                nameText.Size = 13
                nameText.Center = true
                nameText.Outline = true
                Settings.ESP.ActiveDrawings[player] = {Box = box, Name = nameText}
            end
            local v = Settings.ESP.ActiveDrawings[player]
            if onScreen then
                local s = 300 / pos.Z
                local size = Vector2.new(2, 3) * s
                v.Box.Size = size
                v.Box.Position = Vector2.new(pos.X - size.X / 2, pos.Y - size.Y / 2)
                v.Box.Visible = Settings.ESP.Boxes
                v.Name.Text = player.Name
                v.Name.Position = Vector2.new(pos.X, pos.Y - size.Y / 2 - 18)
                v.Name.Visible = Settings.ESP.Names
            else
                v.Box.Visible = false
                v.Name.Visible = false
            end
            if Settings.ESP.Chams then
                if not Settings.ESP.ActiveHighlights[player] or Settings.ESP.ActiveHighlights[player].Parent ~= char then
                    RemoveHighlight(player)
                    local hl = Instance.new("Highlight")
                    hl.FillColor = Color3.fromRGB(150, 40, 255)
                    hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                    hl.FillTransparency = 0.4
                    hl.Adornee = char
                    hl.Parent = char
                    Settings.ESP.ActiveHighlights[player] = hl
                else
                    Settings.ESP.ActiveHighlights[player].Enabled = true
                end
            else
                RemoveHighlight(player)
            end
        else
            if Settings.ESP.ActiveDrawings[player] then
                Settings.ESP.ActiveDrawings[player].Box.Visible = false
                Settings.ESP.ActiveDrawings[player].Name.Visible = false
            end
            RemoveHighlight(player)
        end
    end
end)

local tsSelectedName, tsPlayers, tsCurIdx = "None", {}, 1
local function RefreshTSPlayers()
    tsPlayers = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then table.insert(tsPlayers, p.Name) end
    end
    if #tsPlayers == 0 then
        table.insert(tsPlayers, "None in server")
        tsSelectedName = "None"
        targetStrafeTarget = nil
    elseif not table.find(tsPlayers, tsSelectedName) then
        tsSelectedName = tsPlayers[1]
        targetStrafeTarget = Players:FindFirstChild(tsSelectedName)
    end
end
RefreshTSPlayers()
